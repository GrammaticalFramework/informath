\begin{axiom}\label{felix_informath_forall}
    $\forall x, y. x = x\land y = y$.
\end{axiom}

\begin{axiom}\label{felix_informath_exists}
    $\exists x, y. x = y$.
\end{axiom}

\begin{abbreviation}\label{fixture_regular}
    $A$ is fixture-regular iff $A = A$.
\end{abbreviation}

\begin{proposition}\label{felix_informath_claim}
    Let $A$ be a set.
    Suppose $A$ is fixture-regular.
    Then $A = A$.
\end{proposition}
\begin{proof}
    Follows by \auto[local].
\end{proof}
